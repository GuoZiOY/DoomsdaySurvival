# M4B 系统补全 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 补全新框架的游戏系统，使新技术栈上的游戏完整可玩：训练场/任务板设施（再证「三块积木」可规模化）、区域探索（随机事件）、战斗完整回合（技能/道具/敌人回合/胜利结算渲染）、剧情接通全部特殊目标。

**Architecture:** 沿用 M4 设施框架——新增设施 = facilities.json 定义 + `训练场设施.cs`/`任务板设施.cs` + UXML。探索 = 服务（`探索服务`，权重随机事件，复用 `区域数据`）+ 视图（`探索控制器`）。战斗在 `BattleService` 加技能/道具/敌人回合，`战斗控制器` 渲染消息与技能/道具页。剧情 `学习:`/`任务:` → `设施:`，`探索:`/`区域:` 接探索服务。

**Tech Stack:** C#（`文字RPG.Services` / `文字RPG.Facilities` / `文字RPG.UI`）、JSON（facilities.json 增两条、story.json 局部改）、UI Toolkit。

## Global Constraints

- 设施 = 三块积木，核心零改动；探索/战斗为服务+视图。
- 逻辑永不调 UI；设施/探索发事件，视图订阅。
- 全中文命名 + `//` 行内注释 + 无 `_` 前缀。
- 无自动化测试；验证 = 编译零错误 + Play 观察。
- 主菜单/结局界面、场景切换删旧代码 = M5/M6（本计划不做）。
- story.json 改造暂时破坏旧游戏（可接受，旧代码 M6 删除）。

`训练场.uxml`（标题 + 技能列表 + 返回按钮，结构同商店）。

`训练场设施.cs`：
```csharp
using UnityEngine;
using UnityEngine.UIElements;
using 文字RPG.Events;

namespace 文字RPG.Facilities
{
    // 训练场设施：列出全部技能，点击学习（扣金币、学技能、发日志）
    public sealed class 训练场设施 : 设施
    {
        private VisualElement 技能列表;

        protected override void 绑定视图()
        {
            根.Q<Label>("标题").text = "—— 训练场 ——";
            技能列表 = 根.Q<VisualElement>("技能列表");
            根.Q<Button>("返回按钮").clicked += 返回;
            订阅<金币变化事件>(_ => 重建列表());
            重建列表();
        }

        private void 重建列表()
        {
            技能列表.Clear();
            foreach (var 技能 in 上下文.数据.技能.Values)
            {
                var 模板 = Resources.Load<VisualTreeAsset>("UI/controls/物品条目");
                var 行 = 模板.CloneTree();
                行.Q<Label>("物品名").text = $"{技能.名称}（{技能.描述}）  {技能.价格}金";
                行.Q<Label>("物品数量").text = 上下文.玩家.掌握技能(技能.标识) ? "已学" : "学习";
                var 标识 = 技能.标识;
                行.RegisterCallback<ClickEvent>(_ => 学习(标识));
                技能列表.Add(行);
            }
        }

        private void 学习(string 技能标识)
        {
            if (!上下文.数据.技能.TryGetValue(技能标识, out var 技能)) return;
            var 玩家 = 上下文.玩家;
            if (玩家.掌握技能(技能标识)) { 事件.发布(new 日志事件(文字RPG.Events.日志类型.系统, "你已经掌握这个技能了。")); return; }
            if (玩家.金币 < 技能.价格) { 事件.发布(new 日志事件(文字RPG.Events.日志类型.反馈坏, $"金币不足（需要 {技能.价格}）。")); return; }
            玩家.金币 -= 技能.价格;
            玩家.学习技能(技能标识);
            事件.发布(new 金币变化事件(玩家.金币, -技能.价格));
            事件.发布(new 日志事件(文字RPG.Events.日志类型.反馈, $"你学会了技能：{技能.名称}（-{技能.价格} 金币）"));
        }
    }
}
```

```csharp
using UnityEngine;
using UnityEngine.UIElements;
using 文字RPG.Events;
using 文字RPG.Services;

namespace 文字RPG.Facilities
{
    // 任务板设施：可接任务 + 进行中任务进度
    public sealed class 任务板设施 : 设施
    {
        private VisualElement 任务列表;

        protected override void 绑定视图()
        {
            根.Q<Label>("标题").text = "—— 任务板 ——";
            任务列表 = 根.Q<VisualElement>("任务列表");
            根.Q<Button>("返回按钮").clicked += 返回;
            订阅<任务进度事件>(_ => 重建列表());
            重建列表();
        }

        private void 重建列表()
        {
            任务列表.Clear();
            foreach (var 任务 in 上下文.数据.任务.Values)
            {
                var 模板 = Resources.Load<VisualTreeAsset>("UI/controls/物品条目");
                var 行 = 模板.CloneTree();
                var 玩家 = 上下文.玩家;
                // 已接：显示进度；未接：显示接取
                string 进度文本 = "接取";
                foreach (var 进度 in 玩家.任务)
                    if (进度.标识 == 任务.标识)
                        进度文本 = 进度.已完成 ? "已完成" : $"{进度.数量}/{任务.目标数量}";
                行.Q<Label>("物品名").text = $"{任务.名称}（{任务.描述}）  +{任务.奖励金币}金 +{任务.奖励经验}经验";
                行.Q<Label>("物品数量").text = 进度文本;
                var 标识 = 任务.标识;
                行.RegisterCallback<ClickEvent>(_ => ServiceRegistry.Get<QuestService>().接取(标识));
                任务列表.Add(行);
            }
        }
    }
}
```

- [ ] **Step 1: 写探索服务**

创建 `Assets/Scripts/Services/探索服务.cs`：
```csharp
using System.Collections.Generic;
using UnityEngine;
using 文字RPG.Core;
using 文字RPG.Data;
using 文字RPG.Domain;
using 文字RPG.Events;

namespace 文字RPG.Services
{
    // 探索事件类型
    public enum 探索事件类型 { 遭遇敌人, 发现地点, 获得资源, 无事发生 }

    // 探索服务：进入区域反复深入探索，随机触发事件（权重表）
    public sealed class 探索服务
    {
        private readonly EventBus 事件;
        private readonly DataService 数据;
        private readonly PlayerService 玩家;
        private readonly BattleService 战斗;

        private 区域数据 当前区域;
        public string 返回节点 { get; private set; }
        public string 当前区域标识 { get; private set; }

        public 探索服务(EventBus 事件, DataService 数据, PlayerService 玩家, BattleService 战斗)
        { this.事件 = 事件; this.数据 = 数据; this.玩家 = 玩家; this.战斗 = 战斗; }

        public void 进入区域(string 区域标识, string 返回节点)
        {
            if (!数据.区域.TryGetValue(区域标识, out 当前区域)) { Debug.LogError($"[探索服务] 区域不存在: {区域标识}"); return; }
            this.返回节点 = 返回节点;
            当前区域标识 = 区域标识;
            事件.发布(new 探索显示事件(当前区域.描述 + "\n\n" + 危险度文本(当前区域.危险度), new[] { new 探索选项数据("深入探索", "深入"), new 探索选项数据("返回", "返回") }));
        }

        public void 深入探索() => 执行事件(抽取事件());

        // 权重随机抽取事件（复用旧 区域探索 逻辑）
        private (探索事件类型, string 文本, string 敌人, string 节点, 剧情效果 效果) 抽取事件()
        {
            var 候选 = new List<(探索事件类型, string, string, string, 剧情效果, int)>();
            if (当前区域.遭遇 != null) foreach (var e in 当前区域.遭遇) if (!string.IsNullOrEmpty(e.敌人)) 候选.Add((探索事件类型.遭遇敌人, null, e.敌人, null, null, e.权重));
            if (当前区域.发现 != null) foreach (var d in 当前区域.发现) if (!string.IsNullOrEmpty(d.节点)) 候选.Add((探索事件类型.发现地点, null, null, d.节点, null, d.权重));
            if (当前区域.资源 != null) foreach (var r in 当前区域.资源) if (r.效果 != null) 候选.Add((探索事件类型.获得资源, r.文本, null, null, r.效果, r.权重));
            if (当前区域.无事文本 != null) foreach (var t in 当前区域.无事文本) 候选.Add((探索事件类型.无事发生, t, null, null, null, 1));
            if (候选.Count == 0) return (探索事件类型.无事发生, "这里什么也没有。", null, null, null);
            int 总权重 = 0; foreach (var c in 候选) 总权重 += Mathf.Max(1, c.Item6);
            int 掷点 = Random.Range(0, 总权重);
            foreach (var c in 候选) { 掷点 -= Mathf.Max(1, c.Item6); if (掷点 < 0) return (c.Item1, c.Item2, c.Item3, c.Item4, c.Item5); }
            return (候选[候选.Count - 1].Item1, 候选[候选.Count - 1].Item2, 候选[候选.Count - 1].Item3, 候选[候选.Count - 1].Item4, 候选[候选.Count - 1].Item5);
        }

        private void 执行事件((探索事件类型, string, string, string, 剧情效果) 事件项)
        {
            switch (事件项.Item1)
            {
                case 探索事件类型.遭遇敌人:
                    事件.发布(new 探索显示事件($"你遇上了{数据.敌人[事件项.Item3].名称}！", new[] { new 探索选项数据("进入战斗", "战斗") }));
                    战斗.开始战斗(事件项.Item3, "__探索胜利", "__探索返回");   // M4B 简化：战斗后回探索由控制器处理
                    break;
                case 探索事件类型.发现地点:
                    var 节点 = 事件项.Item4;
                    ServiceRegistry.Get<DialogueService>().进入节点(节点);
                    break;
                case 探索事件类型.获得资源:
                    效果结算.应用(事件, 玩家.档案, 事件项.Item5);
                    事件.发布(new 探索显示事件(事件项.Item2, new[] { new 探索选项数据("继续深入", "深入"), new 探索选项数据("返回", "返回") }));
                    break;
                default:
                    事件.发布(new 探索显示事件(事件项.Item2, new[] { new 探索选项数据("继续深入", "深入"), new 探索选项数据("返回", "返回") }));
                    break;
            }
        }

        private static string 危险度文本(int 危险度)
            => 危险度 switch { 1 => "<color=#7fae6a>危险度：低</color>", 2 => "<color=#d9a441>危险度：中</color>", _ => "<color=#c7473d>危险度：高</color>" };
    }
}
```
> 注：`探索显示事件`/`探索选项数据` 在 `Events/探索事件.cs` 定义（结构同剧情事件）。

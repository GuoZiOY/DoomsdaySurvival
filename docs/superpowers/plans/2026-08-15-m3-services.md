# M3 服务层 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把游戏逻辑做成服务层——玩家服务（PlayerService）、对话引擎（DialogueService）、背包/任务/战斗服务，接入真实玩家档案与剧情 JSON，用事件驱动新 UI（替换 M2 的演示数据）。M3 完成后：新 UI 里可真实推进序章剧情、打怪、做任务，战斗/商店等设施交互留到 M4。

**Architecture:** 服务全部纯逻辑、无 UI 依赖；与 UI 通信仅走事件总线。对话循环用「事件→视图渲染→视图回调服务」闭环：`DialogueService` 进入节点后发布 `显示剧情事件`（携带文本+选项数据）→ `StoryController` 订阅渲染按钮 → 用户点按钮 → `StoryController` 调 `DialogueService.处理选项(目标)` → 服务推进剧情再发事件。玩家数据由 `PlayerService` 持有（创建/读档），并在装配时向 `玩家档案` 注入装备解析器（接 DataService）。旧代码照常运行，不动。

**Tech Stack:** C#（`文字RPG.*` 命名空间，全中文命名、`//` 行内注释）、JSON（story/enemies/items/quests）、事件总线。

## Global Constraints

- 服务在 `文字RPG.Services`；对话/UI 事件在 `文字RPG.Events`；领域复用 `文字RPG.Domain`（玩家档案/战斗规则/效果结算）。
- **逻辑永不调 UI**：服务只发事件；StoryController 等 UI 只订阅 + 回调服务公开方法。
- 选项数据用纯数据载体（`剧情选项数据`），不携带委托——回调由 UI 层转成 `DialogueService.处理选项(目标)`。
- 复用 M1 `效果结算`、`战斗规则`、M2 `StoryController`/`HudController`/`ViewManager`。
- 全中文命名 + `//` 行内注释 + 无 `_` 前缀。
- 无自动化测试；验证 = 编译零错误 + Play 观察（控制台无新框架报错；旧游戏既有报错忽略）。
- 演示数据（`演示数据.cs`）在 M3 完成后移除，由 PlayerService + DialogueService 替代。

```csharp
using 文字RPG.Core;
using 文字RPG.Data;
using 文字RPG.Domain;
using 文字RPG.Events;
using 文字RPG.Services;

namespace 文字RPG.Services
{
    // 玩家服务：持有玩家档案，处理新游戏/读档，注入装备解析，发布初始状态事件
    public sealed class PlayerService
    {
        private readonly EventBus 事件;
        private readonly DataService 数据;
        private readonly SaveService 存档;

        public 玩家档案 档案 { get; private set; }

        public PlayerService(EventBus 事件, DataService 数据, SaveService 存档)
        {
            this.事件 = 事件;
            this.数据 = 数据;
            this.存档 = 存档;
        }

        // 创建并装配玩家档案（装备加成解析器接 DataService）
        public void 初始化()
        {
            档案 = new 玩家档案();
            档案.武器攻击解析 = 标识 => 数据.物品.TryGetValue(标识, out var 物品) ? 物品.攻击加成 : 0;
            档案.防具防御解析 = 标识 => 数据.物品.TryGetValue(标识, out var 物品) ? 物品.防御加成 : 0;
            发布初始状态();
        }

        // 新游戏：重置档案 + 初始物品
        public void 新游戏()
        {
            初始化();
            档案.添加物品("面包", 1);
            档案.当前节点 = "序章_醒来";
            事件.发布(new 日志事件(文字RPG.Events.日志类型.系统, "新的旅程开始了。"));
            发布初始状态();
        }

        // 读档：成功则用存档档案，否则新游戏
        public void 读档()
        {
            var 数据 = 存档.读取();
            if (数据?.玩家 != null)
            {
                档案 = 数据.玩家;
                档案.武器攻击解析 = 标识 => this.数据.物品.TryGetValue(标识, out var 物品) ? 物品.攻击加成 : 0;
                档案.防具防御解析 = 标识 => this.数据.物品.TryGetValue(标识, out var 物品) ? 物品.防御加成 : 0;
                if (档案.生命 <= 0) 档案.生命 = 档案.最大生命;   // 防死档
                事件.发布(new 日志事件(文字RPG.Events.日志类型.系统, "读取存档，继续冒险。"));
            }
            else { 新游戏(); }
            发布初始状态();
        }

        // 发布当前生命/魔力/金币，让 HUD 初始化
        private void 发布初始状态()
        {
            事件.发布(new 生命变化事件(档案.生命, 档案.最大生命, 0));
            事件.发布(new 魔力变化事件(档案.魔力, 档案.最大魔力, 0));
            事件.发布(new 金币变化事件(档案.金币, 0));
        }
    }
}
```

- [ ] **Step 2: 写 DialogueService**

创建 `Assets/Scripts/Services/DialogueService.cs`：
```csharp
using System.Collections.Generic;
using UnityEngine;
using 文字RPG.Core;
using 文字RPG.Data;
using 文字RPG.Domain;
using 文字RPG.Events;

namespace 文字RPG.Services
{
    // 对话引擎：数据驱动推进剧情。进入节点 → 应用效果 → 发布 显示剧情事件；
    // 用户点选项 → StoryController 回调 处理选项(目标) → 推进。
    public sealed class DialogueService
    {
        private readonly EventBus 事件;
        private readonly DataService 数据;
        private readonly PlayerService 玩家;

        private 剧情节点 当前节点;

        // 当前显示缓存：StoryController 绑定时先拉取，避免错过装配期的初始事件
        public 显示剧情事件 当前显示 { get; private set; }

        public DialogueService(EventBus 事件, DataService 数据, PlayerService 玩家)
        {
            this.事件 = 事件;
            this.数据 = 数据;
            this.玩家 = 玩家;
        }

        public void 进入节点(string 节点标识)
        {
            if (!数据.剧情.TryGetValue(节点标识, out var 节点))
            {
                Debug.LogError($"[DialogueService] 节点不存在: {节点标识}");
                return;
            }
            当前节点 = 节点;
            玩家.档案.当前节点 = 节点标识;

            // 进入效果（生命/金币/物品等）
            if (节点.效果 != null)
                效果结算.应用(事件, 玩家.档案, 节点.效果);

            // 构建选项（数据载体，不含委托）
            var 选项列表 = new List<剧情选项数据>();
            if (节点.选项 != null)
                foreach (var 选项 in 节点.选项)
                {
                    // 条件分支：需持有物品才显示
                    if (!string.IsNullOrEmpty(选项.需要物品) && !玩家.档案.持有物品(选项.需要物品)) continue;
                    选项列表.Add(new 剧情选项数据(选项.文本, 选项.目标));
                }

            // 缓存并发布显示事件，StoryController 订阅渲染
            当前显示 = new 显示剧情事件(节点.文本 ?? "", 选项列表.ToArray());
            事件.发布(当前显示);
        }

        // 处理选项目标：节点跳转 / __结束 / 战斗 / 设施 / 探索 等
        public void 处理选项(string 目标)
        {
            if (目标 == "__结束") { 事件.发布(new 日志事件(文字RPG.Events.日志类型.系统, "序章结束。")); return; }
            if (目标.StartsWith("战斗:")) { var p = 目标.Split(':'); if (p.Length >= 3) 事件.发布(new 战斗结束事件(true, p[1])); 进入节点(p.Length >= 3 ? p[2] : 当前节点.标识); return; }
            if (目标.StartsWith("设施:")) { 事件.发布(new 日志事件(文字RPG.Events.日志类型.系统, $"设施[{目标.Substring(3)}] 待 M4 实现")); return; }
            if (目标.StartsWith("探索:")) { 事件.发布(new 日志事件(文字RPG.Events.日志类型.系统, $"探索[{目标}] 待 M4 实现")); return; }
            if (目标.StartsWith("任务:")) { 事件.发布(new 日志事件(文字RPG.Events.日志类型.系统, $"任务[{目标}] 待 M4 实现")); return; }
            进入节点(目标);
        }
    }
}
```

```csharp
public sealed class StoryController : ViewController
{
    private Label 正文;
    private VisualElement 选项区;

    protected override void 绑定视图()
    {
        正文 = 根.Q<Label>("正文");
        选项区 = 根.Q<VisualElement>("选项区");

        // 订阅显示剧情事件：对话引擎 → 渲染
        订阅<显示剧情事件>(显示剧情);

        // 绑定时先拉取当前显示（装配期的事件可能已发出，缓存兜底）
        var 当前 = ServiceRegistry.Get<DialogueService>().当前显示;
        if (当前.文本 != null) 显示剧情(当前);
    }

    private void 显示剧情(显示剧情事件 e)
    {
        正文.text = e.文本;
        选项区.Clear();
        if (e.选项 == null) return;
        foreach (var 选项 in e.选项)
        {
            var 模板 = Resources.Load<VisualTreeAsset>("UI/controls/按钮");
            if (模板 == null) { Debug.LogError("[StoryController] 找不到按钮模板"); return; }
            var 按钮 = 模板.CloneTree();
            按钮.Q<Label>("文字").text = 选项.文本;
            UI反馈.挂载按钮反馈(按钮);
            var 目标 = 选项.目标;   // 捕获副本，避免闭包问题
            按钮.RegisterCallback<ClickEvent>(_ => ServiceRegistry.Get<DialogueService>().处理选项(目标));
            选项区.Add(按钮);
        }
    }
}
```

创建 `Assets/Scripts/Services/QuestService.cs`：
```csharp
using System.Collections.Generic;
using 文字RPG.Core;
using 文字RPG.Data;
using 文字RPG.Domain;
using 文字RPG.Events;

namespace 文字RPG.Services
{
    // 任务服务：接取 / 记录击败 / 记录获得；完成时发放奖励并发 任务完成事件
    public sealed class QuestService
    {
        private readonly EventBus 事件;
        private readonly DataService 数据;
        private readonly 玩家档案 档案;

        public QuestService(EventBus 事件, DataService 数据, 玩家档案 档案) { this.事件 = 事件; this.数据 = 数据; this.档案 = 档案; }

        public bool 接取(string 任务标识)
        {
            if (!数据.任务.TryGetValue(任务标识, out var 任务)) return false;
            if (!档案.添加任务(任务标识)) return false;
            事件.发布(new 任务进度事件(任务标识, 0, 任务.目标数量, false));
            return true;
        }

        public void 记录击败(string 敌人标识)
        {
            foreach (var 进度 in 档案.任务)
            {
                if (进度.已完成 || !数据.任务.TryGetValue(进度.标识, out var 任务)) continue;
                if (任务.目标类型 != "击败" || 任务.目标标识 != 敌人标识) continue;
                进度.数量++;
                if (进度.数量 >= 任务.目标数量) { 进度.已完成 = true; 发放奖励(进度.标识, 任务); }
                事件.发布(new 任务进度事件(进度.标识, 进度.数量, 任务.目标数量, 进度.已完成));
            }
        }

        public void 记录获得(string 物品标识)
        {
            foreach (var 进度 in 档案.任务)
            {
                if (进度.已完成 || !数据.任务.TryGetValue(进度.标识, out var 任务)) continue;
                if (任务.目标类型 != "获得物品" || 任务.目标标识 != 物品标识) continue;
                进度.数量++;
                if (进度.数量 >= 任务.目标数量) { 进度.已完成 = true; 发放奖励(进度.标识, 任务); }
                事件.发布(new 任务进度事件(进度.标识, 进度.数量, 任务.目标数量, 进度.已完成));
            }
        }

        private void 发放奖励(string 任务标识, 任务数据 任务)
        {
            档案.金币 += 任务.奖励金币;
            if (!string.IsNullOrEmpty(任务.奖励物品)) 档案.添加物品(任务.奖励物品);
            bool 升级了 = 档案.获得经验(任务.奖励经验);
            事件.发布(new 金币变化事件(档案.金币, 任务.奖励金币));
            事件.发布(new 任务完成事件(任务标识, $"{任务.名称} 完成！+{任务.奖励金币}金 +{任务.奖励经验}经验"));
            if (升级了) 事件.发布(new 经验变化事件(档案.等级, 档案.经验, 档案.升级所需经验, true));
        }
    }
}
```

```csharp
namespace 文字RPG.Services

    // 战斗服务：回合制逻辑（复用 战斗规则 纯函数）。UI 交互面板 M4 做，本任务先有服务逻辑。
    public sealed class BattleService
    {
        private readonly EventBus 事件;
        private readonly DataService 数据;
        private readonly 玩家档案 档案;

        private 敌人数据 敌人;
        private int 敌人生命;
        private bool 防御中;
        private string 胜利节点, 返回节点;

        public BattleService(EventBus 事件, DataService 数据, 玩家档案 档案) { this.事件 = 事件; this.数据 = 数据; this.档案 = 档案; }

        public bool 战斗中 { get; private set; }

        public void 开始战斗(string 敌人标识, string 胜利后节点, string 返回节点)
        {
            if (!数据.敌人.TryGetValue(敌人标识, out 敌人)) { Debug.LogError($"[BattleService] 敌人不存在: {敌人标识}"); return; }
            this.胜利节点 = 胜利后节点; this.返回节点 = 返回节点;
            敌人生命 = 敌人.生命; 防御中 = false; 战斗中 = true;
            事件.发布(new 日志事件(文字RPG.Events.日志类型.操作, $"{敌人.名称} 出现了！"));
        }

        public void 玩家攻击() => 敌人受击(战斗规则.普通伤害(档案.总攻击, 敌人.防御, Random.Range(-1, 2)), "你挥剑攻击！");

        public void 玩家技能(string 技能标识)
        {
            if (!数据.技能.TryGetValue(技能标识, out var 技能)) return;
            if (!档案.消耗魔力(技能.消耗魔力)) { 事件.发布(new 日志事件(文字RPG.Events.日志类型.反馈坏, "魔力不足。")); return; }
            事件.发布(new 魔力变化事件(档案.魔力, 档案.最大魔力, -技能.消耗魔力));
            switch (技能.类型)
            {
                case "重击": 敌人受击(战斗规则.技能伤害(档案.总攻击, 技能.数值, 敌人.防御, Random.Range(-1, 2)), $"你发动{技能.名称}！"); break;
                case "防御": 防御中 = true; 事件.发布(new 日志事件(文字RPG.Events.日志类型.操作, "你摆出防御架势。")); break;
                case "回血": 档案.恢复生命(技能.数值); 事件.发布(new 生命变化事件(档案.生命, 档案.最大生命, 技能.数值)); break;
            }
            敌人回合();
        }

        private void 敌人受击(int 伤害, string 前缀)
        {
            敌人生命 = Mathf.Max(0, 敌人生命 - 伤害);
            事件.发布(new 日志事件(文字RPG.Events.日志类型.操作, $"{前缀}造成 {伤害} 点伤害。"));
            if (敌人生命 <= 0) { 胜利(); return; }
            敌人回合();
        }

        private void 敌人回合()
        {
            int 伤害 = 战斗规则.普通伤害(敌人.攻击, 档案.总防御, Random.Range(-1, 2));
            if (防御中) 伤害 = 战斗规则.防御后伤害(伤害);
            档案.受到伤害(伤害);
            事件.发布(new 生命变化事件(档案.生命, 档案.最大生命, -伤害));
            事件.发布(new 日志事件(文字RPG.Events.日志类型.反馈坏, $"{敌人.名称} 对你造成 {伤害} 点伤害。"));
            if (档案.生命 <= 0) { 战败(); return; }
            防御中 = false;
        }

        private void 胜利()
        {
            战斗中 = false;
            档案.金币 += 敌人.金币奖励;
            档案.获得经验(敌人.经验奖励);
            事件.发布(new 金币变化事件(档案.金币, 敌人.金币奖励));
            事件.发布(new 战斗结束事件(true, 敌人.标识));
            // 掉落
            if (!string.IsNullOrEmpty(敌人.掉落物品) && Random.value < 敌人.掉落概率)
            { 档案.添加物品(敌人.掉落物品); 事件.发布(new 背包变化事件(敌人.掉落物品, 1, 变化原因.获得)); }
        }

        private void 战败() { 战斗中 = false; 事件.发布(new 战斗结束事件(false, 敌人.标识)); }
    }
}
```
